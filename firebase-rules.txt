// ANTES de publicar (senão você se tranca para fora):

{
  "rules": {
    // Lista de acesso: ninguém lê nem altera pelo app (só pelo console)
    "allowedUsers": {
      ".read": false,
      ".write": false
    },
    "patients": {
      ".read": "auth != null && root.child('allowedUsers').child(auth.uid).val() === true",
      ".write": "auth != null && root.child('allowedUsers').child(auth.uid).val() === true",
      ".indexOn": ["createdAt", "lastContactAt"],
      "$patientId": {
        // Paciente novo precisa de nome e situação; exclusão (null) é permitida
        ".validate": "!newData.exists() || newData.hasChildren(['name', 'status'])",
        "name": {
          ".validate": "newData.isString() && newData.val().length > 0 && newData.val().length <= 120"
        },
        "phone": {
          ".validate": "newData.isString() && newData.val().matches(/^[0-9]{0,15}$/)"
        },
        "status": {
          ".validate": "newData.val() === 'Paciente solicitado risco' || newData.val() === 'Paciente agendou cirurgia' || newData.val() === 'Paciente fez cirurgia' || newData.val() === 'Cirurgias Refrativas' || newData.val() === 'Paciente não quer operar'"
        },
        "notes": {
          ".validate": "newData.isString() && newData.val().length <= 5000"
        }
      }
    },
    // Pedidos de lentes de contato (contact-lenses.js) — mesma equipe autorizada
    "contactLensOrders": {
      ".read": "auth != null && root.child('allowedUsers').child(auth.uid).val() === true",
      ".write": "auth != null && root.child('allowedUsers').child(auth.uid).val() === true",
      ".indexOn": ["createdAt", "status"],
      "$orderId": {
        // Pedido precisa de paciente, situação e data; exclusão (null) é permitida
        ".validate": "!newData.exists() || newData.hasChildren(['patientName', 'status', 'orderDate'])",
        "patientName": {
          ".validate": "newData.isString() && newData.val().length > 0 && newData.val().length <= 120"
        },
        "phone": {
          ".validate": "newData.isString() && newData.val().matches(/^[0-9]{0,15}$/)"
        },
        "status": {
          ".validate": "newData.val() === 'ordered' || newData.val() === 'arrived' || newData.val() === 'notified' || newData.val() === 'delivered' || newData.val() === 'cancelled'"
        },
        "orderDate": {
          ".validate": "newData.isString() && newData.val().matches(/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/)"
        },
        "expectedDate": {
          ".validate": "newData.isString() && newData.val().matches(/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/)"
        },
        "brand": { ".validate": "newData.isString() && newData.val().length <= 80" },
        "supplier": { ".validate": "newData.isString() && newData.val().length <= 80" },
        "notes": { ".validate": "newData.isString() && newData.val().length <= 2000" },
        "boxes": { ".validate": "newData.isNumber() && newData.val() >= 1 && newData.val() <= 99" },
        "quantity": { ".validate": "newData.isNumber() && newData.val() >= 1 && newData.val() <= 99" },
        "quantityUnit": { ".validate": "newData.val() === 'caixa' || newData.val() === 'par_teste'" },
        "durationMonths": { ".validate": "newData.isNumber() && newData.val() >= 1 && newData.val() <= 24" },
        "price": { ".validate": "newData.isNumber() && newData.val() >= 0" }
      }
    }
  }
}
